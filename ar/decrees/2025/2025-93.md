# مـــرســـوم رقـــم (93) لـسـنـــة 2025 بـالـتـصـديــق عـلـى اتـفـاقـيـة تـعـاون بـين حـكـومـة دولـة قـطـر وحـكـومـة جـمـهـوريـة بـنـين فـي مـجـال الـتـعـلـيـم والـتـعـلـيـم الـعـالـي والـبـحـث الـعـلـمـي

**نـحـن تمـيـم بـن حـمــد آل ثـانـي أمـيـر دولـة قـطـــر ،**

بـعـد الاطــلاع عـلـى الـدسـتـور،

وعـلـى وثـيـقـة الـتـصـديـق الـصــادرة فـي الــسـادس والـعـشـريـن مـن شـهــر ذي الـحـجـة عـام 1446 هــجـريـة ، المـوافــق لـلـثـانـي والـعـشـريـن من شـهـر يـونـيـو عـام 2025 مـيـلاديـة ،

وعـلـى اقـتـراح رئـيـس مـجـلـس الـوزراء ووزيـر الخـارجـيـة ،

وعـلـى مـشـروع المـرسـوم المـقـدم مـن مـجـلـس الـوزراء ،

**رسمنـا بمـا هـو آت :**

## المواد (1-2)

### مادة 1

صُـودق عـلـى اتـفـاقـيـة تـعـاون بـين حـكـومـة دولـة قـطـر وحـكـومـة جـمـهـوريـة بـنـين فـي مـجـال الـتـعـلـيـم والـتـعـلـيـم الـعـالـي والـبـحـث الـعـلـمـي ، المـوقـعـة بمـديـنـة الـدوحـة بــتـاريـخ 20 /05/ 2025 ، المـرفـق نـصـهـا بـهـذا المـرسـوم ، وتـكـون لـهـا قـوة الـقـانـون ، وفـقـاً للـمـادة (68) مـن الـدسـتـور .

### مادة 2

عـلـى جـمـيـع الجـهـات المـخـتـصـة ، كـل فـيـمـا يـخـصـه ، تـنـفـيـذ هـذا المـرســوم . ويُـعـمـل بـه مـن تـاريـخ صــدوره . ويُـنـشــر فـي الجـريــدة الـرسـمـيـة .

تميم بن حمد آل ثاني — أمير دولة قطر
