# مـــرســـوم رقـــم (102) لـسـنـــة 2025 بـالـتـصـديــق عـلـى اتـفـاقـيـة بـشـأن تـشـجـيـع وحـمـايـة الاسـتـثـمـارات المـتـبـادلـة بـين حـكـومـة دولـة قـطـر وحـكـومـة جـمـهـوريـة روانـدا

**نـحـن تمـيـم بـن حـمــد آل ثـانـي أمـيـر دولـة قـطـــر ،**

بـعـد الاطــلاع عـلـى الـدسـتـور،

وعـلـى وثـيـقـة الـتـصـديـق الـصــادرة فـي الـثـانـي والـعـشـريـن مـن شـهــر ذي الـقـعـدة عـام 1441 هــجـريـة ، المـوافــق للـثـالـث عـشـر من شـهـر يـولـيـو عـام 2020 مـيـلاديـة ،

وعـلـى اقـتـراح رئـيـس مـجـلـس الـوزراء ووزيـر الخـارجـيـة ،

وعـلـى مـشـروع المـرسـوم المـقـدم مـن مـجـلـس الـوزراء ،

**رسمنـا بمـا هـو آت :**

## المواد (1-2)

### مادة 1

صُـودق عـلـى اتـفـاقـيـة بـشـأن تـشـجـيـع وحـمـايـة الاسـتـثـمـارات المـتـبـادلـة بـين حـكـومـة دولـة قـطـر وحـكـومـة جـمـهـوريـة روانـدا ، المـوقـعـة بمـديـنـة الـدوحـة بــتـاريـخ 15/ 11/ 2018 ، المـرفـق نـصـهـا بـهـذا المـرسـوم ، وتـكـون لـهـا قـوة الـقـانـون ، وفـقـاً لـلـمـادة (68) مـن الـدسـتـور .

### مادة 2

عـلـى جـمـيـع الجـهـات المـخـتـصـة ، كـل فـيـمـا يـخـصـه ، تـنـفـيـذ هـذا المـرســوم . ويُـعـمـل بـه مـن تـاريـخ صــدوره . ويُـنـشــر فـي الجـريــدة الـرسـمـيـة .

تميم بن حمد آل ثاني — أمير دولة قطر
