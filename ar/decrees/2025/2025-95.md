# مــرســوم رقــم (95) لـسـنـة 2025 بـالـتـصـديـق عـلـى بـروتـوكـول بـتـعـديـل الاتـفـاقـيـة بـين حـكـومـة دولــة قــطــر وحـكـومـة ممـلـكـة الـنـرويـج بشـأن تـجـنـب الازدواج الـضـريـبـي ومـنـع الـتـهـرب المـالـي فـيـمـا يـتـعـلـق بالـضـرائـب عـلـى الـدخـل مـع الـبـروتـوكـول المـوقـعـين فـي أوسـلـو فـي 29 يـونـيـو 2009

**نـحـن تمـيـم بـن حـمــد آل ثـانـي أمـيـر دولـة قـطــر ،**

بـعـد الاطـلاع عـلـى الـدستـور،

وعـلى وثـيـقـة الـتـصـديـق الــصــادرة فــي الــسـابــع مــن شــهــر جــمــادى الآخـــرة عـــام 1446 هـجـريـة ، المـوافـق لـلـثـامـن مــن شــهــر ديـسـمـبـر عـــام 2024 مـيـلاديـة ،

وعـلى اقـتــراح رئيـس مجـلــس الــوزراء ووزيـــر الخـــارجــية ،

وعـلى مـشـــروع المــرســـوم المـقـــدم مـن مجـلـس الــوزراء ،

**رسـمنــا بمـا هــو آت :**

## المواد (1-2)

### مادة 1

صُـودق عـلـى بـروتـوكـول بـتـعـديـل الاتـفـاقـيـة بـين حـكـومـة دولــة قــطــر وحـكـومـة ممـلـكـة الـنـرويـج بشـأن تـجـنـب الازدواج الـضـريـبـي ومـنـع الـتـهـرب المـالـي فـيـمـا يـتـعـلـق بالـضـرائـب عـلـى الـدخـل مـع الـبـروتـوكـول المـوقـعـين فـي أوسـلـو فـي 29 يـونـيـو 2009 ، المـوقـع بمــدينــة أوسـلـو بتــاريـخ 04/ 09/ 2024 ، المـرفـق نـصــه بـهـذا المـرسـوم ، وتــكــون لــه قــوة الـقـانـون ، وفـقـاً لـلـمـادة (68) مـن الدسـتـور .

### مادة 2

عـلـى جـمـيـع الجـهـات المـخـتـصـة ، كـل فـيـمـا يـخـصـه ، تـنـفـيـذ هـذا المـرســوم . ويُـعـمـل بـه مـن تـاريـخ صــدوره . ويُـنـشــر فـي الجـريــدة الـرسـمـيـة .

تميم بن حمد آل ثاني — أمير دولة قطر
